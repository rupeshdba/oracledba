#Query to check last modified time of object
set lines 300
col object_name for a38
col object_type for a23
col owner for a20
col status for a19
select owner,object_name,to_char(last_ddl_time,'dd-mm-yyyy hh24:mi:ss') Last_DDL_Time from dba_objects where object_name='&object_name' and owner='&USER';


#Query to check when the object was created
set lines 300
col object_name for a38
col object_type for a23
col owner for a20
col status for a19
select owner,object_name,to_char(created,'dd-mm-yyyy hh24:mi:ss') Created from dba_objects where object_name='&object_name' and owner='&USER';
