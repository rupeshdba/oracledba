set lines 300 pages 3000
col owner for a23
col object_type for a33
col object_name for a36
select owner,object_name,object_type,to_char(created,'dd-mm-yyyy hh24:mi:ss') created,to_char(last_ddl_time,'dd-mm-yyyy hh24:mi:ss') last_ddl_time from dba_objects where status='INVALID' order by 1;
