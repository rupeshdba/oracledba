#How to find if database is Primary or Standby in Oracle

SQL> select name,open_mode,controlfile_type,database_role from v$database;

NAME      OPEN_MODE            CONTROL DATABASE_ROLE
--------- -------------------- ------- ----------------
PLUGDB    READ WRITE           CURRENT PRIMARY


SQL> select name,open_mode,database_role from v$database;

NAME      OPEN_MODE            DATABASE_ROLE
--------- -------------------- ----------------
PR        MOUNTED              PHYSICAL STANDBY
