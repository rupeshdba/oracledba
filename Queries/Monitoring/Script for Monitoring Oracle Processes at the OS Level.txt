#Login by oracle user and copy this script in a location other than ORACLE_HOME. Ensure the directory is owned by oracle:oinstall with execute permission. Suppose, the directory name is /backup.

vi /backup/session_count.sh
while true
do
date
echo "The LOCAL=NO count is: `ps -ef|grep -i LOCAL=NO|wc -l` "
ps -ef|grep -i LOCAL=NO

echo "The LOCAL=YES count is: `ps -ef|grep -i LOCAL=YES|wc -l` "
ps -ef|grep -i LOCAL=YES

echo ""
echo ""
sleep 5
done


$cd /backup

#Command to run the script.
$nohup sh session_count.sh > session_count.log &
